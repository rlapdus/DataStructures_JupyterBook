Reverse: Stack¶

  1. Create a function that reverses individual characters in a stack. Note that the original stack should be reversed.

  2. Add a reverse() method to the ListStack class.

def reverse(stack):
    temp = []  # 임시 리스트

    # 다 꺼내서 temp에 저장
    while not stack.isEmpty():
        temp.append(stack.pop())

    # 다시 넣기
    for item in temp:
        stack.push(item)

stack = ListStack()
stack.push(100); stack.push(200); stack.push(300); stack.push(400)

class ListStack:
    def __init__(self):
        self.stack = []

    def push(self, x):
        self.stack.append(x)

    def pop(self):
        return self.stack.pop()

    def isEmpty(self):
        return len(self.stack) == 0

    def printStack(self):
        print(self.stack)

    def reverse(self):
        temp = []


        while not self.isEmpty():
            temp.append(self.pop())


        for item in temp:
            self.push(item)

def reverseQueue(queue):
    if queue.isEmpty():
        return
    temp = queue.dequeue()
    reverseQueue(queue)
    queue.enqueue(temp)

def reverse(self):
    if self.isEmpty():
        return

    temp = self.dequeue()
    self.reverse()
    self.enqueue(temp)

Palindrome¶

  • Palindrome is word, phrase, or sequence that reads the same backward as forward (e.g. madam, mom, kayak). Write a function that checks if a string is palindrome using stack and queue data structure.

from DataStructureCode.ListStack import *
from DataStructureCode.ListQueue import *

def isPalindrome(A):
    stack = ListStack()
    queue = ListQueue()

    for ch in A:
        stack.push(ch)
        queue.enqueue(ch)

    while not queue.isEmpty():
        if stack.pop() != queue.dequeue():
            return False

    return True

str = 'civic'
print(str, "is Palindrome?: ", isPalindrome(str))
civic is Palindrome?:  True
